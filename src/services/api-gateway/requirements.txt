# JWT authentication -- BOTH libraries are genuinely needed here, unlike every
# other service (#797): PyJWT for shared.auth.jwt_middleware's own HS256
# session JWTs (same as everyone else), but python-jose stays too because
# core/oidc.py's Authelia ID-token verification relies on python-jose's
# built-in OIDC `at_hash` claim check (binds the ID token to its paired
# access token) -- PyJWT has no equivalent (confirmed: zero references to
# at_hash anywhere in its source). Dropping python-jose here would either
# silently lose that check or require hand-rolling OIDC at_hash verification,
# which is its own decision, not bundled into this dependency swap.
PyJWT[crypto]==2.13.0
python-jose[cryptography]==3.5.0
python-multipart==0.0.32

# Password hashing (core/auth.py imports bcrypt's own top-level API directly --
# checkpw/gensalt/hashpw -- not passlib's wrapper; passlib[bcrypt] was only ever
# pulling this in transitively, #341)
